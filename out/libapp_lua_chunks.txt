utils = {}
function utils.md5 (input)
    return vmplugin.invoke_method('utils_md5', input)
end
function utils.to_upper (input)
    return vmplugin.invoke_method('utils_upper_case', input)
end
function utils.to_lower (input)
    return vmplugin.invoke_method('utils_lower_case', input)
end
function utils.timestamp ()
    return vmplugin.invoke_method('utils_timestamp', '')
end
function utils.base64_encode (input, option)
    option = option or {}
    if (option["mode"] == nil) then option["mode"] = "utf8" end
    if (option["mode"] == "byte") then
      local bytes = {}
      for i=1,#input,1 do
          bytes[i] = string.byte(input,i)
      end
      return vmplugin.invoke_method('utils_base64_encode', json.encode({text = bytes, mode = option["mode"]}))
    end
    return vmplugin.invoke_method('utils_base64_encode', json.encode({text = input, mode = option["mode"]}))
end
function utils.base64_decode (input, option)
    option = option or {}
    if (option["mode"] == nil) then option["mode"] = "utf8" end
    local ret = vmplugin.invoke_method('utils_base64_decode', json.encode({text = input, mode = option["mode"]}))
    if (option["mode"] == "byte") then
      local array = json.decode(ret)
      local str = ""
      for i,c in ipairs(array) do
        if( c ~= nil ) then
            str = str..string.char(c)
        end
      end
      return str
    end
    return ret 
end
function utils.aes128cbc_encrypt (key, iv, data, option)
    option = option or {}
    local key_array = {}
    for i=1,#key,1 do
        key_array[i] = string.byte(key,i)
    end
    local iv_array = {}
    for i=1,#iv,1 do
        iv_array[i] = string.byte(iv,i)
    end
    return vmplugin.invoke_method('utils_aes128cbc_encrypt', json.encode({key = key_array, iv = iv_array, data = data, padding = option["padding"]}))
end
function utils.aes128cbc_decrypt (key, iv, data, option)
    option = option or {}
    local key_array = {}
    for i=1,#key,1 do
        key_array[i] = string.byte(key,i)
    end
    local iv_array = {}
    for i=1,#iv,1 do
        iv_array[i] = string.byte(iv,i)
    end
    return vmplugin.invoke_method('utils_aes128cbc_decrypt', json.encode({key = key_array, iv = iv_array,  data = data, padding = option["padding"]}))
end
